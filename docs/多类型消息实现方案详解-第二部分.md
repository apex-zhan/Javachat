#### **Step 5：创建Handler**

```java
package com.abin.mallchat.common.chat.service.strategy.msg;

import com.abin.mallchat.common.chat.dao.MessageDao;
import com.abin.mallchat.common.chat.domain.entity.Message;
import com.abin.mallchat.common.chat.domain.entity.msg.LocationMsgDTO;
import com.abin.mallchat.common.chat.domain.entity.msg.MessageExtra;
import com.abin.mallchat.common.chat.domain.enums.MessageTypeEnum;
import com.abin.mallchat.common.chat.domain.vo.request.msg.LocationMsgReq;
import com.abin.mallchat.common.common.utils.AssertUtil;
import org.springframework.beans.BeanUtils;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Component;

import java.util.Optional;

/**
 * 位置消息处理器
 */
@Component
public class LocationMsgHandler extends AbstractMsgHandler<LocationMsgReq> {
    
    @Autowired
    private MessageDao messageDao;

    @Override
    MessageTypeEnum getMsgTypeEnum() {
        return MessageTypeEnum.LOCATION;
    }

    /**
     * 扩展校验：位置坐标合法性
     */
    @Override
    protected void checkMsg(LocationMsgReq body, Long roomId, Long uid) {
        // 1. 校验纬度范围 [-90, 90]
        AssertUtil.isTrue(
            body.getLatitude() >= -90 && body.getLatitude() <= 90,
            "纬度范围错误"
        );
        
        // 2. 校验经度范围 [-180, 180]
        AssertUtil.isTrue(
            body.getLongitude() >= -180 && body.getLongitude() <= 180,
            "经度范围错误"
        );
    }

    /**
     * 保存位置消息
     */
    @Override
    protected void saveMsg(Message msg, LocationMsgReq body) {
        MessageExtra extra = Optional.ofNullable(msg.getExtra())
            .orElse(new MessageExtra());
        
        // 转换为DTO
        LocationMsgDTO dto = LocationMsgDTO.builder()
            .latitude(body.getLatitude())
            .longitude(body.getLongitude())
            .description(body.getDescription())
            .address(body.getAddress())
            .build();
        
        // 可选：生成地图快照（调用第三方地图API）
        // dto.setMapSnapshotUrl(mapService.generateSnapshot(dto));
        
        Message update = new Message();
        update.setId(msg.getId());
        update.setContent(body.getDescription());  // content存储位置描述
        update.setExtra(extra);
        
        extra.setLocationMsgDTO(dto);
        
        messageDao.updateById(update);
    }

    /**
     * 展示位置消息
     */
    @Override
    public Object showMsg(Message msg) {
        return msg.getExtra().getLocationMsgDTO();
    }

    /**
     * 回复时展示
     */
    @Override
    public Object showReplyMsg(Message msg) {
        LocationMsgDTO location = msg.getExtra().getLocationMsgDTO();
        return "[位置]" + location.getDescription();
    }

    /**
     * 会话列表展示
     */
    @Override
    public String showContactMsg(Message msg) {
        return "[位置]" + msg.getContent();
    }
}
```

#### **5.2.1 Controller层**

```java
@RestController
@RequestMapping("/capi/chat")
public class ChatController {
    @Autowired
    private ChatService chatService;

    /**
     * 发送消息
     */
    @PostMapping("/msg")
    @FrequencyControl(time = 5, count = 3, target = FrequencyControl.Target.UID)
    public ApiResult<ChatMessageResp> sendMsg(@Valid @RequestBody ChatMessageReq request) {
        // 获取当前用户
        Long uid = RequestHolder.get().getUid();
        
        // 调用Service
        Long msgId = chatService.sendMsg(request, uid);
        
        // 查询消息详情
        ChatMessageResp resp = chatService.getMsgResp(msgId, uid);
        
        return ApiResult.success(resp);
    }
}
```

**关键点：**
- `@FrequencyControl`：频控注解，防止刷消息（5秒内最多3条）
- `@Valid`：参数校验，触发JSR303校验
- `RequestHolder.get().getUid()`：从ThreadLocal获取当前用户

#### **5.2.2 Service层**

```java
@Service
public class ChatServiceImpl implements ChatService {
    @Autowired
    private RoomService roomService;
    @Autowired
    private ApplicationEventPublisher applicationEventPublisher;

    @Override
    @Transactional(rollbackFor = Exception.class)
    public Long sendMsg(ChatMessageReq request, Long uid) {
        // 1. 校验房间是否存在
        check(request, uid);
        
        // 2. 获取Handler处理消息
        AbstractMsgHandler msgHandler = MsgHandlerFactory.getStrategyNoNull(request.getMsgType());
        Long msgId = msgHandler.checkAndSaveMsg(request, uid);
        
        // 3. 发布消息事件（事务提交后触发）
        applicationEventPublisher.publishEvent(new MessageSendEvent(this, msgId));
        
        return msgId;
    }

    private void check(ChatMessageReq request, Long uid) {
        // 校验房间是否存在
        Room room = roomService.getById(request.getRoomId());
        AssertUtil.isNotEmpty(room, "房间不存在");
        
        // 校验是否在群内（群聊需要）
        if (room.isGroupRoom()) {
            RoomGroup roomGroup = roomService.getRoomGroup(room.getId());
            GroupMember member = groupMemberDao.getMember(roomGroup.getId(), uid);
            AssertUtil.isNotEmpty(member, "您不在群聊中");
        }
        
        // 校验黑名单
        // ...
    }
}
```

**关键点：**
- `@Transactional`：保证消息保存和事件发布在同一事务
- `ApplicationEventPublisher`：Spring事件发布器
- `MessageSendEvent`：自定义事件

### 5.3 核心技术点分析

#### **5.3.1 泛型的妙用**

```java
public abstract class AbstractMsgHandler<Req> {
    private Class<Req> bodyClass;

    @PostConstruct
    private void init() {
        // 通过反射获取泛型类型
        ParameterizedType genericSuperclass = 
            (ParameterizedType) this.getClass().getGenericSuperclass();
        this.bodyClass = (Class<Req>) genericSuperclass.getActualTypeArguments()[0];
        
        // 自动注册到工厂
        MsgHandlerFactory.register(getMsgTypeEnum().getType(), this);
    }

    private Req toBean(Object body) {
        // 利用泛型类型进行转换
        if (bodyClass.isAssignableFrom(body.getClass())) {
            return (Req) body;
        }
        return BeanUtil.toBean(body, bodyClass);
    }
}
```

**工作原理：**

```
TextMsgHandler extends AbstractMsgHandler<TextMsgReq>
                                           ↑
                                           │
                      泛型实际类型：TextMsgReq.class
                                           │
                                           ↓
          通过反射获取：getActualTypeArguments()[0]
                                           │
                                           ↓
                     赋值给 bodyClass 字段
                                           │
                                           ↓
                      用于 toBean() 类型转换
```

**优势：**
- ✅ 类型安全：编译期检查
- ✅ 自动转换：无需手动强转
- ✅ 代码简洁：避免大量instanceof判断

#### **5.3.3 Spring的@PostConstruct自动注册**

```java
public abstract class AbstractMsgHandler<Req> {
```

**执行时机：**

```

   ↓
```

**优势：**
- ✅ 无需手动配置
- ✅ 启动时注册，运行时无性能损耗
- ✅ 支持热插拔（添加新Handler自动注册）

```java
// 保存消息时计算间隔
if (Objects.nonNull(body.getReplyMsgId())) {

        msg.getRoomId(), 
        body.getReplyMsgId(), 
        msg.getId()
    );
    update.setGapCount(gapCount);
    update.setReplyMsgId(body.getReplyMsgId());
}

SELECT COUNT(*)
```

**展示时判断是否可跳转：**

### 6.2 实战案例：AI生成消息

#### **需求：**
集成ChatGPT，用户可以@机器人，机器人回复AI生成的消息。

#### **实现步骤：**

**Step 1：定义AI消息类型**
```java
public enum MessageTypeEnum {
    // ... 现有类型
    AI_GENERATED(10, "AI生成消息"),
}
```

**Step 2：创建AIMsgHandler**
```java
@Component
public class AIMsgHandler extends AbstractMsgHandler<AIMsgReq> {
    @Autowired
    private ChatGPTService chatGPTService;
    
    @Override
    MessageTypeEnum getMsgTypeEnum() {
        return MessageTypeEnum.AI_GENERATED;
    }
    
    @Override
    protected void saveMsg(Message msg, AIMsgReq body) {
        // 1. 调用ChatGPT API
        String aiResponse = chatGPTService.chat(body.getPrompt());
        
        // 2. 保存AI响应
        MessageExtra extra = Optional.ofNullable(msg.getExtra())
            .orElse(new MessageExtra());
        Message update = new Message();
        update.setId(msg.getId());
        update.setContent(aiResponse);
        update.setExtra(extra);
        
        extra.setAiMsgDTO(AIMsgDTO.builder()
            .prompt(body.getPrompt())
            .model("gpt-3.5-turbo")
            .tokens(aiResponse.length())
            .build());
        
        messageDao.updateById(update);
    }
    
    @Override
    public Object showMsg(Message msg) {
        return AIMsgResp.builder()
            .content(msg.getContent())
            .prompt(msg.getExtra().getAiMsgDTO().getPrompt())
            .model(msg.getExtra().getAiMsgDTO().getModel())
            .build();
    }
}
```

**Step 3：TextMsgHandler中触发AI消息**

### 6.3 性能优化建议

#### **6.3.1 批量查询优化**

```java
// ❌ N+1查询问题
for (Message msg : messages) {
    User user = userDao.getById(msg.getFromUid());  // N次查询
}

// ✅ 批量查询
List<Long> userIds = messages.stream()

    .collect(Collectors.toList());
Map<Long, User> userMap = userInfoCache.getBatch(userIds);  // 1次查询

for (Message msg : messages) {
    User user = userMap.get(msg.getFromUid());
}
```

#### **6.3.2 缓存优化**

```java
@Component
public class MsgCache {
    @Autowired
    private MessageDao messageDao;
    
    // Caffeine本地缓存
    private final Cache<Long, Message> cache = Caffeine.newBuilder()
        .maximumSize(10000)
        .expireAfterWrite(Duration.ofMinutes(5))
        .build();
    
    public Message getMsg(Long msgId) {
        return cache.get(msgId, messageDao::getById);
    }
}
```
